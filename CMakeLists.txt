cmake_minimum_required(VERSION 3.24)
project(GModPort LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GMOD_BUILD_ENGINE "Build all Source modules and the iOS application" ${IOS})
include(CTest)
add_library(port_module_registry STATIC platform/static_modules.cpp)
target_include_directories(port_module_registry PUBLIC platform)
if(BUILD_TESTING AND NOT IOS)
  add_executable(static_modules_test tests/static_modules_test.cpp)
  target_link_libraries(static_modules_test PRIVATE port_module_registry)
  if(MSVC)
    target_compile_options(static_modules_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(static_modules_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME static_module_registry COMMAND static_modules_test)
endif()
if(GMOD_BUILD_ENGINE)
  if(NOT IOS)
    message(FATAL_ERROR "The engine target currently requires CMAKE_SYSTEM_NAME=iOS")
  endif()
  enable_language(OBJC OBJCXX)
  include(cmake/SourceEngine.cmake)
endif()
