cmake_minimum_required(VERSION 3.24)
project(GModPort LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GMOD_BUILD_ENGINE "Build all Source modules and the iOS application" ${IOS})
include(CTest)
add_library(port_module_registry STATIC platform/static_modules.cpp)
target_include_directories(port_module_registry PUBLIC platform)
if(BUILD_TESTING AND NOT IOS)
  add_executable(static_modules_test tests/static_modules_test.cpp)
  target_link_libraries(static_modules_test PRIVATE port_module_registry)
  if(MSVC)
    target_compile_options(static_modules_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(static_modules_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME static_module_registry COMMAND static_modules_test)
  set(SOURCE_ENGINE_HEADERS "${CMAKE_SOURCE_DIR}/../ref/source-engine" CACHE PATH "Pinned Source headers for factory integration tests")
  if(EXISTS "${SOURCE_ENGINE_HEADERS}/public/tier1/interface.h")
    set(factory_test_sources tests/source_loader_test.cpp platform/source_loader.cpp)
    foreach(module engine client)
      if(module STREQUAL "engine")
        set(FACTORY_VALUE 11)
      else()
        set(FACTORY_VALUE 22)
      endif()
      set(prefix "${CMAKE_CURRENT_BINARY_DIR}/${module}_test_prefix.h")
      file(WRITE "${prefix}" "#define InterfaceReg gp_test_${module}_InterfaceReg\n#define CreateInterface gp_test_${module}_CreateInterface\n#define CreateInterfaceInternal gp_test_${module}_CreateInterfaceInternal\n#define Sys_GetFactoryThis gp_test_${module}_Sys_GetFactoryThis\n")
      configure_file(tests/factory_registration.cpp.in "${module}_test_registration.cpp" @ONLY)
      configure_file(platform/module_factory.cpp.in "${module}_test_factory.cpp" COPYONLY)
      set(module_sources "${CMAKE_CURRENT_BINARY_DIR}/${module}_test_registration.cpp" "${CMAKE_CURRENT_BINARY_DIR}/${module}_test_factory.cpp")
      if(MSVC)
        set_source_files_properties(${module_sources} PROPERTIES COMPILE_OPTIONS "/FI${prefix}")
      else()
        set_source_files_properties(${module_sources} PROPERTIES COMPILE_OPTIONS "-include;${prefix}")
      endif()
      list(APPEND factory_test_sources ${module_sources})
    endforeach()
    add_executable(source_loader_test ${factory_test_sources})
    target_include_directories(source_loader_test PRIVATE "${SOURCE_ENGINE_HEADERS}/public" "${SOURCE_ENGINE_HEADERS}/public/tier0")
    target_link_libraries(source_loader_test PRIVATE port_module_registry)
    if(MSVC)
      target_compile_options(source_loader_test PRIVATE /UNDEBUG)
      target_compile_definitions(source_loader_test PRIVATE COMPILER_MSVC=1)
    else()
      target_compile_options(source_loader_test PRIVATE -UNDEBUG)
      target_compile_definitions(source_loader_test PRIVATE POSIX=1 OSX=1 GNUC=1)
    endif()
    add_test(NAME source_interface_registration COMMAND source_loader_test)
  endif()
endif()
if(GMOD_BUILD_ENGINE)
  if(NOT IOS)
    message(FATAL_ERROR "The engine target currently requires CMAKE_SYSTEM_NAME=iOS")
  endif()
  enable_language(OBJC OBJCXX)
  include(cmake/SourceEngine.cmake)
endif()
