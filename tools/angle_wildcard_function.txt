bool NamesMatchWithWildcard(const std::string_view &glob, const std::string_view &name)
{
    // GModPort: bounded indices instead of raw string_view pointer arithmetic.
    size_t g = 0, n = 0, star = std::string_view::npos, retry = 0;
    while (n < name.size())
    {
        if (g < glob.size() && glob[g] != '*' && (glob[g] == '?' || glob[g] == name[n]))
        {
            ++g;
            ++n;
        }
        else if (g < glob.size() && glob[g] == '*')
        {
            star = g++;
            retry = n;
        }
        else if (star != std::string_view::npos && retry < name.size())
        {
            g = star + 1;
            n = ++retry;
        }
        else
        {
            return false;
        }
    }
    while (g < glob.size() && glob[g] == '*') ++g;
    return g == glob.size();
}
